04/

十題 JavaScript 基礎綜合練習，涵蓋 if、for、while、function、JSON、array、object。

// 01 — if/function
function checkScore(score) {
  if (score < 0 || score > 100) return "錯誤輸入";
  if (score >= 90) return "A";
  if (score >= 80) return "B";
  if (score >= 70) return "C";
  return "F";
}
console.log(checkScore(85));  // B
// 02 — for/function
function sumRange(n) {
  let total = 0;
  for (let i = 1; i <= n; i++) total += i;
  return total;
}
console.log(sumRange(10));  // 55
// 03 — while/array
const nums = [3, 7, 2, 9, 5];
let index = 0;
while (index < nums.length) {
  if (nums[index] > 8) {
    console.log(`找到數字 ${nums[index]} 在索引 ${index}`);
    break;
  }
  index++;
}
// 08 — for/object/JSON (購物車結帳)
const jsonList = '[{"name": "書", "price": 200}, {"name": "筆", "price": 30}]';
const items = JSON.parse(jsonList);
let total = 0;
for (let item of items) total += item.price;
console.log(`總金額為: ${total}`);  // 總金額為: 230
// 10 — object/function (銀行提款)
let bankAccount = {
  balance: 1000,
  withdraw(amount) {
    if (this.balance >= amount) {
      this.balance -= amount;
      console.log(`提款成功，餘額：${this.balance}`);
    } else { console.log("餘額不足"); }
  }
};
bankAccount.withdraw(400);  // 提款成功，餘額：600
bankAccount.withdraw(800);  // 餘額不足

JavaScript 進階概念練習

06_new/

十題進階練習，涵蓋 Callback、IIFE、map、箭頭函式、自訂 filter、setTimeout、reduce 等。

// 01 — Callback 函式
function mathTool(num1, num2, action) {
  return action(num1, num2);
}
console.log(mathTool(10, 5, function(a, b) { return a + b; }));  // 15
// 02 — IIFE
(function() {
  let count = 100;
  console.log("Count is: " + count);
})();
// 03 — map()
const prices = [100, 200, 300, 400];
const discounted = prices.map(p => p * 0.8);
console.log(discounted);  // [80, 160, 240, 320]
// 05 — 高階函式
const multiplier = (factor) => (n) => n * factor;
const double = multiplier(2);
console.log(double(10));  // 20
// 10 — reduce + Callback
function calculateTotal(cart, discountFunc) {
  const total = cart.reduce((sum, price) => sum + price, 0);
  return discountFunc(total);
}
console.log(calculateTotal([100, 200, 300], total => total - 50));  // 550
